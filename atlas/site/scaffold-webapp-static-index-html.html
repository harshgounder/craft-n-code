<script>
const $=id=>document.getElementById(id);
let FEED={items:[]};

async function jget(u){const r=await fetch(u);return r.json()}

async function load(){
  const [feed,digest,stats]=await Promise.all([
    jget('/api/feed'),jget('/api/digest'),jget('/api/stats')]);
  FEED=feed;
  $('mode-chip').textContent='mode: '+ (stats.mode||'live');
  $('llm-chip').textContent='llm: '+ (feed.llm?.model||'db');
  $('n-chip').textContent=feed.total+' items';
  renderStats(stats);
  renderDigest(digest);
  renderFeed(feed.items);
  renderComplaints((await jget('/api/complaints')).complaints);
  renderTrace((await jget('/api/trace')).steps||[]);
}

function renderStats(s){
  $('stats').innerHTML=[
    ['channels',s.channels?Object.keys(s.channels).length:'-'],
    ['deadlines found',s.deadlines_found],
    ['urgent now',FEED.items.filter(i=>i.is_urgent).length],
    ['skin',s.skin_ready?'mount-ready':'-'],
  ].map(([l,n])=>`<div class="stat"><div class="n">${n}</div><div class="l">${l}</div></div>`).join('');
}

function renderDigest(d){
  const ur=d.urgent||[];
  $('digest').innerHTML = ur.length? ur.map(i=>`
    <div class="urg-item"><span class="tag">URGENT</span>
      <div class="txt">${esc(i.summary)}
        ${i.deadline_iso?`<div class="dl">⏰ deadline ${i.deadline_iso}</div>`:''}
      </div></div>`).join('')
    : '<div class="txt" style="color:var(--mut)">nothing urgent - you are clear today</div>';
}

const CH_ICON={email:'📧',portal:'🏛',chat:'💬',app:'📱',social:'📸',ticket:'🛠'};
function renderFeed(items){
  $('feed').innerHTML=items.map((i,k)=>`
    <div class="card ${i.is_urgent?'urgent':''}">
      <div class="rank">#${k+1}<small>${i.rank_score}</small></div>
      <div style="flex:1">
        <span class="chan">${CH_ICON[i.channel]||''} ${i.channel}</span>
        <h3>${esc(i.subject)}</h3>
        <div class="sum">${esc(i.summary)}</div>
        <div class="meta">
          <span class="sender">${esc(i.sender)}</span>
          ${i.deadline_iso?`<span class="dl">⏰ ${i.deadline_iso}</span>`:''}
          ${i.is_urgent?'<span style="color:var(--warn);font-weight:700">⚠ urgent</span>':''}
        </div>
      </div>
    </div>`).join('');
}

async function doSearch(){
  const q=$('q').value.trim();
  if(!q)return;
  const d=await jget('/api/search?q='+encodeURIComponent(q));
  $('search-res').innerHTML=d.results.length?
    d.results.map((i,k)=>`
      <div class="card"><div class="rank">#${k+1}</div>
        <div style="flex:1"><span class="chan">${i.channel}</span>
        <h3>${esc(i.subject)}</h3>
        <div class="sum">${esc(i.summary)}</div>
        <div class="meta">${i.deadline_iso?`<span class="dl">⏰ ${i.deadline_iso}</span>`:''}</div></div>
      </div>`).join('')
    : '<div style="color:var(--mut);padding:20px">no match - try "mte", "fee", "lab"…</div>';
}

const SLA=[['open',1.0],['in-progress',.6],['fixed',0]];
async function renderComplaints(list){
  $('complaints').innerHTML=list.map((c,i)=>{
    const status=c.status||'open';
    const pct=(SLA.find(s=>s[0]===status)||SLA[0])[1];
    return `<div class="card">
      <div class="rank">#C-${114+i}</div>
      <div style="flex:1"><span class="chan">🛠 ${c.category||'general'}</span>
        <h3>${esc(c.subject)}</h3>
        <div class="sum">${esc(c.body)}</div>
        <div class="slabar"><span>${status}</span><div class="bar"><span style="width:${pct*100}%"></span></div>
        <span>SLA ${c.sla_hours||48}h · escalated after ${(c.sla_hours||48)*2}h</span></div>
      </div></div>`;
  }).join('');
}

async function fileComplaint(){
  const title=$('c-title').value.trim(), body=$('c-body').value.trim();
  if(!title){$('c-status').textContent='say what is broken first';return}
  const r=await fetch('/api/complaints',{method:'POST',headers:{'Content-Type':'application/json'},
    body:JSON.stringify({title,body,category:$('c-cat').value,sender:'You (demo)'})});
  const d=await r.json();
  $('c-status').textContent=`${d.ticket} filed → auto-triaged to ${d.item.category} · SLA 48h · you get pinged on status change`;
  $('c-title').value='';$('c-body').value='';
  renderComplaints((await jget('/api/complaints')).complaints);
}

function esc(s){return (s||'').replace(/[&<>"]/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]))}

function stepSummary(st){
  if(st.step==='ingest')return `received ${st.n} items from all channels`;
  if(st.step==='dedupe')return `dropped ${st.dropped} near-duplicates, kept ${st.kept}`;
  if(st.step==='summarize')return `summarized ${(st.per_item||[]).length} items in ${st.mode} mode`;
  if(st.step==='rank')return `ranked; top 3 shown below`;
  return '';
}

function renderTrace(steps){
  $('trace-body').innerHTML=steps.map(st=>`
    <div class="tstep">
      <div class="ts-name">${esc(st.step)} <span style="color:var(--mut);font-weight:400">· ${esc(st.ts||'')}</span></div>
      <div class="ts-sum">${esc(stepSummary(st))}</div>
      <pre>${esc(JSON.stringify(st,null,2))}</pre>
    </div>`).join('')||'<div style="color:var(--mut)">no trace yet</div>';
}

function toggleTrace(){
  const b=$('trace-body'),c=$('trace-caret');
  const open=b.classList.toggle('hide');
  c.classList.toggle('open',!open);
}

const STATUS_CLS={pending:'pending',executed:'executed',approved:'approved',rejected:'rejected',snoozed:'snoozed'};
function renderProposals(list){
  $('proposals').innerHTML=list.map(p=>{
    const cls=STATUS_CLS[p.status]||'pending';
    const disabled=p.status==='executed'||p.status==='approved'||p.status==='rejected'||p.status==='snoozed';
    const ev=(p.evidence||[]).map(e=>
      `<div class="ev"><span class="ev-src">${esc(e.source_id)}</span>
        <div class="ev-body">${esc(e.snippet)}</div></div>`).join('');
    const params=JSON.stringify(p.params||{});
    return `<div class="prop" id="prop-${p.id}">
      <h3>${esc(p.tool)} <span class="conf">${(p.confidence*100).toFixed(0)}%</span>
        <span class="chip status ${cls}">${p.status}</span></h3>
      <div class="reason">${esc(p.reason)}</div>
      <div class="params">${esc(params)}</div>
      <div class="ev" style="cursor:pointer" onclick="this.style.display='none';$('ev-${p.id}').style.display='block'">
        evidence (click to expand)</div>
      <div class="ev hide" id="ev-${p.id}">${ev}</div>
      <div class="btns">
        <button class="pbtn approve" ${disabled?'disabled':''} onclick="decide('${p.id}','approve')">APPROVE</button>
        <button class="pbtn reject" ${disabled?'disabled':''} onclick="decide('${p.id}','reject')">REJECT</button>
        <button class="pbtn snooze" ${disabled?'disabled':''} onclick="decide('${p.id}','snooze')">SNOOZE</button>
      </div>
    </div>`;
  }).join('')||'<div style="color:var(--mut);padding:20px">no proposals yet - propose a tool from the feed</div>';
}

function renderAudit(events){
  $('audit').innerHTML=events.map(e=>`
    <tr><td class="dt">${esc(e.ts)}</td><td>${esc(e.actor)}</td>
      <td>${esc(e.decision)}</td><td>${esc(e.tool)}</td></tr>`).join('')||
    '<tr><td colspan="4" style="color:var(--mut)">no audit events yet</td></tr>';
}

async function loadActions(){
  renderProposals((await jget('/api/proposals')).proposals);
  renderAudit((await jget('/api/audit')).events);
}

async function decide(id,decision){
  const r=await fetch('/api/approve',{method:'POST',headers:{'Content-Type':'application/json'},
    body:JSON.stringify({proposal_id:id,decision,actor:'user'})});
  const d=await r.json();
  if(d.ok){loadActions();}
}

document.querySelectorAll('.tab').forEach(t=>t.onclick=()=>{
  document.querySelectorAll('.tab').forEach(x=>x.classList.remove('on'));
  t.classList.add('on');
  ['feed','search','complaints','actions'].forEach(v=>$('v-'+v).classList.toggle('hide',v!==t.dataset.v));
  if(t.dataset.v==='actions')loadActions();
});

load();
</script>
</body>
